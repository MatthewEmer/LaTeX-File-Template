\newcommand{\newchapter}[1]{\chapter{#1}\label{chapter:#1}} % #1 Chapter name.

\newcommand{\newsubsection}[1]{\subsection{#1}\label{subsection:#1}} % #1 Subsection name.

\newcommand{\insertfigure}[4]{ % #1 Label, #2 Image source, #3 Image width, #4 Caption.
    \begin{figure}[H]
        \centering
        \includegraphics[width=#3\linewidth]{#2}
        \caption{#4}\label{figure:#1}
    \end{figure}
}

\newcommand{\currenttablelabel}{}
\newcommand{\currenttablecaption}{}
\counterwithout{table}{chapter}
\newenvironment{Table}[3]{ % #1 Label, #2 Column layout, #3 Caption.
    \renewcommand{\currenttablelabel}{#1}
    \renewcommand{\currenttablecaption}{#3}
    \begin{table}[H]
        \centering
        \begin{tabular}{#2}
}{%
        \end{tabular}
        \caption{\currenttablecaption}\label{table:\currenttablelabel}
    \end{table}
}

\newenvironment{Pseudocode}[4]{ % #1 Label, #2 Inputs, #3 Outputs, #4 Caption.
    \begin{algorithm}[H]
        \caption{#4}\label{algorithm:#1}
        \begin{algorithmic}
            \State \textbf{Inputs}: #2
            \State \textbf{Outputs}: #3
            \separate{5}
}{%
        \end{algorithmic}
    \end{algorithm}
}

\newenvironment{Snippet}[2]{ % #1 Label, #2 Caption.
    \refstepcounter{algorithm}\label{algorithm:#1}
    \par\addvspace{0.8\baselineskip}
    \hrule

    \separate{2}\textbf{Algorithm \thealgorithm} #2 

    \separate{2}\hrule
}{
    \hrule\par\addvspace{0.8\baselineskip}
}
